# #505 boot 2: the guest sees the mkdisk-created raw image as a 1 GiB disk and writes a marker.
timeout 480000
expect localhost login:
send root\n
expect ~#
# 1 GiB = 2097152 sectors; the marker carries the guest's OWN reading of the size.
send echo QSIZE-$(cat /sys/block/vda/size)\n
expect QSIZE-2097152
send printf 'HYPE505MARKER' | dd of=/dev/vda bs=512 seek=2048 conv=fsync 2>/dev/null && echo WR-OK\n
expect WR-OK
send dd if=/dev/vda bs=512 skip=2048 count=1 2>/dev/null | head -c 13\n
expect HYPE505MARKER
pass raw-marker-written
