# #167: drive FreeBSD 15.0's bsdinstall from the stock disc1 ISO to a finished install, over the
# serial console. Run by tools/167/run-167.sh, which gives the guest a 4 GB file-backed disk.
#
# What this script encodes, each learned from a failed run:
#
# * Input goes over SERIAL (`send`), not the keyboard. FreeBSD starts its primary installer on
#   ttyu0 ("Dual Console: Serial Primary, Video Secondary"); `sendkey` reaches the i8042 and
#   never advances the dialog.
# * Arrow keys are ESC O B / ESC O A, not ESC [ B. The console type answered below is vt100, and
#   ncurses matches the terminfo application-mode sequences. The ESC bytes in this file are raw
#   0x1b: the script language has no escape for it.
# * A button is chosen by its first letter (O = Offline, E = Entire Disk, F = Finish, ...), so
#   no step depends on which button a dialog happens to focus.
# * Install type is PACKAGES, then OFFLINE. This disc1 carries no distribution sets at all --
#   /usr/freebsd-dist holds only MANIFEST -- so "Distribution Sets" always ends in "Network
#   Installation" and, with no NIC, an abort. The offline package set is in
#   /usr/freebsd-packages/offline.
# * A space cannot be sent alone (the parser trims it), so nothing here toggles a checklist.
#
# The package install is ~311 packages and takes about six minutes under nested KVM.
timeout 280000
expect Console type [vt100]:
send \n
timeout 60000
# Not `expect Install`: the "FreeBSD Installer" banner contains it and renders before the dialog.
expect [  Install  ]
delay 2000
send \n
expect Set Hostname
delay 1000
send hype-vm0\n
expect Select Installation Type
delay 1000
send P
expect Network or Offline Installation
delay 1000
send O
# Partitioning: second menu item is Auto (UFS). ZFS is the default and wants more RAM.
expect Partitioning
delay 1500
send OB
delay 1000
send \n
expect Entire Disk
delay 1500
send E
# GPT is the default scheme.
expect Partition Scheme
delay 1500
send \n
expect Finish
delay 3000
send F
expect Commit
delay 1500
send C
# newfs runs here. It issues writes whose segments split a sector across a page boundary, which
# hype's virtio-blk used to refuse (fixed under this ticket) -- armed for the rest of the run.
fail-if Error mounting partition
timeout 120000
expect Select System Components
delay 2000
send \n
timeout 900000
expect Set root password
delay 2000
# Password, Down to the repeat field, password again, Enter.
send hype167OBhype167\n
timeout 60000
expect No network interfaces
delay 2000
send \n
# First region is "0 UTC".
expect Select a region
delay 2000
send \n
expect look reasonable
delay 2000
send Y
# Date, then time: Enter keeps what the guest clock already says.
expect Set Date
delay 2000
send \n
expect Set Time
delay 2000
send \n
# Services and hardening: defaults (sshd, dumpdev; no hardening).
expect System Configuration
delay 2000
send \n
expect System Hardening
delay 2000
send \n
# "bsdconfig firmware" probes for a while, then shows a self-dismissing notice.
timeout 180000
expect Add User Accounts
delay 2000
send N
timeout 60000
expect Final Configuration
delay 2000
send \n
expect Manual Configuration
delay 2000
send N
expect Complete
delay 2000
send R
delay 30000
pass freebsd-install-complete
