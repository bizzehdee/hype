# SMP-8 (#192) end-to-end bar, and the measurement run for #509's bar.
#
# Proves from INSIDE the guest: 2 CPUs online, both schedulable under load,
# reschedule IPIs delivered in BOTH directions, and no lockup/stall over a
# sustained run. The markers are quote-split ("O''K") so the tty echo of the
# typed command cannot satisfy its own expect.
timeout 300000
fail-if soft lockup
fail-if detected stalls
fail-if self-detected stall

expect localhost login:
send root\n
expect ~#

# 2 processors, both online.
send echo NPROC-$(grep -c ^processor /proc/cpuinfo)\n
expect NPROC-2
send echo ONLINE-$(cat /sys/devices/system/cpu/online)\n
expect ONLINE-0-1
send grep -E '^processor|^apicid' /proc/cpuinfo\n
expect ~#

# IPI counters before load, per CPU column.
send grep -E 'RES|CAL|TLB' /proc/interrupts\n
expect ~#
send A=$(awk '/RES/{print $2}' /proc/interrupts)\n
expect ~#
send B=$(awk '/RES/{print $3}' /proc/interrupts)\n
expect ~#

# 3 spinners on 2 CPUs force migrations and preemption on both.
send yes>/dev/null & yes>/dev/null & yes>/dev/null &\n
expect ~#
send sleep 5; killall yes; echo LOAD-DON''E\n
expect LOAD-DONE

# Both CPUs accumulated user time -> both schedulable.
send awk '/^cpu[01] /{print $1"-user-"$2}' /proc/stat\n
expect cpu1-user-

# Reschedule IPIs increased on BOTH CPUs during the load.
send A2=$(awk '/RES/{print $2}' /proc/interrupts)\n
expect ~#
send B2=$(awk '/RES/{print $3}' /proc/interrupts)\n
expect ~#
send [ $A2 -gt $A ] && [ $B2 -gt $B ] && echo RES-IPI-BOTH-O''K\n
expect RES-IPI-BOTH-OK

# Sustained segment: idle a minute, then a second load burst.
delay 60000
send yes>/dev/null & yes>/dev/null & yes>/dev/null &\n
expect ~#
send sleep 10; killall yes; echo LOAD2-DON''E\n
expect LOAD2-DONE
delay 30000

# Final counters for the record, then the lockup/stall verdict from dmesg.
# Two traps cost a passing run each: bare 'stall' substring-matches Alpine's own
# "Installing packages" dmesg line, and typing the literal fail-if strings makes
# the tty echo trip the fail-if. Regex '.' keeps the match while the typed text
# never contains the armed patterns.
send grep -E 'RES|CAL|TLB' /proc/interrupts\n
expect ~#
send echo SMP8-DMESG-$(dmesg | grep -icE 'soft.lockup|detected.stall|hung.task|blocked for more')\n
expect SMP8-DMESG-0

pass smp8-alpine-2vcpu
