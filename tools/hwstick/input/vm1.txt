# HW physical-write validation, vm1: a direct-boot Linux (initramfs shell) writes a marker to the
# SPARE NVMe (5ME3N005713803V2W) via hype's physical NVMe backend (presented as virtio-blk /dev/vda,
# BARs programmed by #588) and reads it back byte-exact. Writes auto-confirmed for this serial.
timeout 300000
fail-if Kernel panic
fail-if Unable to mount root
expect job control turned off
send \n
send test -b /dev/vda && echo HAVE-VDA-$?\n
expect HAVE-VDA-0
send printf 'HYPE-PHYS-NVME!!' | dd of=/dev/vda bs=512 conv=fsync 2>/dev/null; echo MARK-$?\n
expect MARK-0
send sync; echo SYNCED\n
expect SYNCED
send echo NVME-$(dd if=/dev/vda bs=16 count=1 2>/dev/null)\n
expect NVME-HYPE-PHYS-NVME!!
pass physical-nvme-write-verified
