# HW physical-write validation, vm0: Alpine live writes to the SPARE SATA SSD (2132E5BF4EAE) via
# hype's physical AHCI backend, presented as /dev/vda. The 48 MiB dd drives the #295 vectored-write
# path (watch hype's `BLK WRITE ... vec=` DIAG); the marker read-back proves the bytes landed on
# the real disk byte-exact. Writes are operator-authorized for this serial (auto-confirmed).
timeout 300000
fail-if Kernel panic
expect localhost login:
send root\n
expect ~#
send ls -l /dev/vda; echo DEVLIST-$?\n
expect DEVLIST-0
# a large sequential write -- gives the guest block layer every chance to merge, so what reaches
# hype reflects a real workload (the #295 measurement).
send dd if=/dev/zero of=/dev/vda bs=1M count=48 conv=fsync; echo DD-DONE\n
expect DD-DONE
# now a distinct marker at LBA0 and a byte-exact read-back -- proves the physical write path.
send printf 'HYPE-PHYS-AHCI!!' | dd of=/dev/vda bs=512 conv=fsync 2>/dev/null; echo MARK-$?\n
expect MARK-0
send sync\n
expect ~#
send echo AHCI-$(dd if=/dev/vda bs=16 count=1 2>/dev/null)\n
expect AHCI-HYPE-PHYS-AHCI!!
pass physical-ahci-write-verified
