# #223: member A. Static address on the switch, then wait to be pinged and ping back.
timeout 480000
expect localhost login:
send root\n
expect ~#
send ip link set eth0 up && ip addr add 10.223.0.1/24 dev eth0 && echo NETUP-A\n
expect NETUP-A
# Give member B time to come up and address itself, then prove A->B works too.
delay 45000
send ping -c 4 -W 3 10.223.0.2 && echo PING-A''B-OK || echo PING-A''B-FAIL\n
fail-if PING-AB-FAIL
expect PING-AB-OK
# The neighbour table must hold B's REAL MAC (bridge semantics), not hype's router MAC
# (52:54:00:FF:...). A router-MAC entry means the switch path silently fell back to routing.
fail-if 52:54:00:ff
send ip neigh show 10.223.0.2 | tr A-Z a-z && echo NEIGH-DON''E\n
expect NEIGH-DONE
pass switch-a-pinged-b-with-real-mac
