# #688 ext4: the guest's virtio-blk disk is a fully allocated image on the drive's ext4
# partition. A signature write, a read-back and a 64 MiB bulk write prove the
# in-place writer on that filesystem; the host-side fsck after the run proves the
# volume survived it.
timeout 1200000
fail-if soft lockup
fail-if READBACK-DIFF

expect localhost login:
send root\n
expect ~#
send echo BLK-$(ls /dev/vd* 2>/dev/null | tr '\n' ' ')\n
expect BLK-
send dd if=/dev/urandom of=/tmp/sig.bin bs=4096 count=1 2>&1; echo SIG-$?\n
expect SIG-
send dd if=/tmp/sig.bin of=/dev/vda bs=4096 count=1 seek=64 oflag=direct 2>&1; echo WRITE-$?\n
expect WRITE-
send sync; dd if=/dev/vda bs=4096 count=1 skip=64 iflag=direct of=/tmp/back.bin 2>&1; echo READ-$?\n
expect READ-
send cmp /tmp/sig.bin /tmp/back.bin && echo READBACK-MATC''H || echo READBACK-DIF''F\n
expect READBACK-MATCH
send dd if=/dev/urandom of=/dev/vda bs=1M count=64 seek=1 oflag=direct 2>&1; echo BULK-$?\n
expect BULK-0
send sync; echo EXT4-WRITE-DON''E\n
expect EXT4-WRITE-DONE
pass EXT4-WRITE
