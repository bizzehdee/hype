# hype3b.cfg (#688+#689): write-test pass across all three partitions of the
# USB-SATA drive in one boot. Mirrors docs/hw-val-386.md's per-format shape:
# FAT32 write-past-EOF, ext4 write-into-a-hole, NTFS in-place write.
#
# DEVICE NAMES: /dev/vda1, /dev/vda2, /dev/vda3.
#   - hype3b.cfg now attaches the drive WHOLE (partition = whole) as the VM's
#     single physical disk, because hype presents at most one physical-backed
#     disk per VM -- the earlier three-[disk.*] form silently dropped ext4 and
#     NTFS (boot/main.c's extra-slot loop: "only the FIRST attached disk may
#     be physical -- NOT presented");
#   - no `bus` + os_hint = linux -> virtio-blk, slot 0 -> vda;
#   - whole-disk exposure keeps the real partition table, so the three
#     partitions appear as vda1/vda2/vda3.
# The probe line prints what actually appeared first; if it shows sd* instead,
# the front-end resolved to AHCI and every mount below needs sda1..3.
timeout 900000
fail-if soft lockup

expect localhost login:
send root\n
expect ~#
send echo BLK-$(ls /dev/vd* /dev/sd* 2>/dev/null | tr '\n' ' ')\n
expect BLK-
send modprobe ntfs3 2>&1; modprobe vfat 2>&1; echo MOD-$?\n
expect MOD-
send mkdir -p /mnt/fat32 /mnt/ext4 /mnt/ntfs\n
expect ~#
send mount /dev/vda1 /mnt/fat32 2>&1; echo MOUNTFAT-$?\n
expect MOUNTFAT-
send mount /dev/vda2 /mnt/ext4 2>&1; echo MOUNTEXT4-$?\n
expect MOUNTEXT4-
send mount /dev/vda3 /mnt/ntfs 2>&1; echo MOUNTNTFS-$?\n
expect MOUNTNTFS-
send dd if=/dev/urandom of=/mnt/fat32/past-eof.bin bs=4096 count=1 seek=1024 2>&1; echo FATWRITE-$?\n
expect FATWRITE-
send dd if=/dev/urandom of=/mnt/ext4/hole.bin bs=1M count=1 seek=64 2>&1; echo EXT4WRITE-$?\n
expect EXT4WRITE-
send dd if=/dev/urandom of=/mnt/ntfs/inplace.bin bs=4096 count=16 2>&1; echo NTFSWRITE1-$?\n
expect NTFSWRITE1-
send dd if=/dev/urandom of=/mnt/ntfs/inplace.bin bs=4096 count=4 conv=notrunc 2>&1; echo NTFSWRITE2-$?\n
expect NTFSWRITE2-
send sync && umount /mnt/fat32 /mnt/ext4 /mnt/ntfs; echo DONE-$?\n
expect DONE-
pass usbsata-triple-write
