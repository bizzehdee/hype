# hype3c.cfg vm0 (run3c-nvme): sustained dd write workload against the
# physical NVMe scratch target, concurrent with vm1's AHCI workload --
# #713 needs the stall pattern to actually RECUR (its own ~6-7s baseline
# persisting after an initial 46s stall), so this loops rather than
# writing once. Same dd bs=1M shape as tools/295/tools/715's own precedent.
#
# DEVICE NAME: /dev/nvme0n1, not vdb. [disk.nvmescratch] sets `bus = nvme`
# explicitly, and an explicit bus on the VM's first attached disk wins
# (boot/main.c fw_1_target_bus, #202 slice 6d) -- hype presents a real NVMe
# controller at PCI dev 5 and the guest binds it as nvme0n1. This matches
# tools/input-scripts/nvme519-vm0.txt, which is the proven precedent.
# WHOLE DISK, destructive by design: this is a sanctioned scratch drive and
# the writes go straight at the block device, wiping its partition table.
timeout 2700000
fail-if soft lockup

expect localhost login:
send root\n
expect ~#
send echo BLK-$(ls /dev/nvme0n1 /dev/vd* /dev/sd* 2>/dev/null | tr '\n' ' ')\n
expect BLK-
send i=0; while [ $i -lt 40 ]; do dd if=/dev/urandom of=/dev/nvme0n1 bs=1M count=32 seek=$((i*32)) oflag=direct 2>&1 | tail -1; i=$((i+1)); done; echo NVME-LOOP-DONE\n
expect NVME-LOOP-DONE
pass nvme-dd-write
