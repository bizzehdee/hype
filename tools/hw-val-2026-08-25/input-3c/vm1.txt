# hype3c.cfg vm1 (run3c-ahci): sustained dd write workload against the
# physical AHCI scratch target, concurrent with vm0's NVMe workload -- see
# nvme-dd-write.txt's own comment for why this loops.
#
# DEVICE NAME: /dev/sda, not vdb. [disk.ahciscratch] sets `bus = ahci-sata`
# explicitly and it is the VM's first attached disk, so hype presents a real
# ICH9 AHCI HBA with the plain-ATA signature and libata claims it as sda.
# The install ISO is a SEPARATE ATAPI device on its own HBA and becomes sr0,
# so it does not take the sda name -- but the probe below prints the actual
# list first, because that ordering is the one assumption here worth checking.
# WHOLE DISK, destructive by design: sanctioned scratch drive, writes go
# straight at the block device.
timeout 2700000
fail-if soft lockup

expect localhost login:
send root\n
expect ~#
send echo BLK-$(ls /dev/sd* /dev/vd* /dev/sr* 2>/dev/null | tr '\n' ' ')\n
expect BLK-
send i=0; while [ $i -lt 40 ]; do dd if=/dev/urandom of=/dev/sda bs=1M count=32 seek=$((i*32)) oflag=direct 2>&1 | tail -1; i=$((i+1)); done; echo AHCI-LOOP-DONE\n
expect AHCI-LOOP-DONE
pass ahci-dd-write
